342! Calculate the minimum coarse grid cell size as implied by the cell area
343min_area_fv3 = 1.e+14
344do j = 1,dimy_fv3
345 do i = 1,dimx_fv3
346 min_area_fv3 = min(min_area_fv3,area_fv3(i,j))
347 end do
348end do
349! The square root of min_area_FV3 will count as the minimum cell size
350min_dx = sqrt(min_area_fv3)/1000._real_kind ! grid size in km
351! NOTE: min_DX will be used after the big loop below to determine whether
352! to copy topographic statistics to "large-scale" file
353
354
355
356
357!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!
358! This is a loop over all the FV3 (coarse) grid cells
359! The subgrid-scale topographic variables needed for the large-scale
360! orographic gravity wave drag schemes are calculated by the following steps:
361! 1) Sample the fine-scale (30sec) topography contained within each
362! coarse grid cell.
363! 2) Calculate the orographic statistics: stddev,convexity,oa1,...oa4,
364! ol1,...,ol4
365!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!
366
367!$OMP PARALLEL DO DEFAULT(SHARED) PRIVATE(I,DLTA_LAT,DLTA_LON) &
368!$OMP PRIVATE(I_BLK,J_BLK,S_II,S_JJ,E_II,E_JJ,LON_BLK,LAT_BLK,II_M,JJ_M) &
369!$OMP PRIVATE(ZS,II,JJ,II_LOC,JJ_LOC,SUM2,NFINEPOINTS,CELL_COUNT,ZS_ACCUM,ZS_MEAN) &
370!$OMP PRIVATE(SUM4,VAR,NU,ND,RATIO,NW,NT)
371do j = 1,dimy_fv3
372 do i = 1,dimx_fv3
373
374 cell_count = ( (j-1) * dimx_fv3 ) + i
375
376 ! Calculate approximate side-lengths of square lat-long "coarse" grid
377 ! cell centered on FV3 cell (units = radians)
378 dlta_lat = sqrt(area_fv3(i,j))/ae
379 dlta_lon = sqrt(area_fv3(i,j))/(ae*cos(lat_fv3(i,j)))
380
381 ! Determine lat/lon of 9 lat-lon block centers
382 ! Note: lat_blk(2)/lon_blk(2) = lat_FV3(i,j)/lon_FV3(i,j)
383 ! Note: abs(lon_blk) may exceed pi
384 do i_blk = 1,3
385 lon_blk(i_blk) = lon_fv3(i,j) + (i_blk-2)*dlta_lon
386 end do
387 ! Note: abs(lat_blk) may exceed pi/2 (90 degrees)
388 do j_blk = 1,3
389 lat_blk(j_blk) = lat_fv3(i,j) + (j_blk-2)*dlta_lat
390 end do
391
392 ! Find starting and ending fine-grid i,j indices for each
393 ! of the 9 "coarse-grid" blocks
394 ! Note: Index value of -999 is returned if latitude of grid points
395 ! exceed 90 degrees north or south
396 do i_blk = 1,3
397 s_ii(i_blk) = nearest_i_east(lon_blk(i_blk)-p5*dlta_lon)
398 e_ii(i_blk) = nearest_i_west(lon_blk(i_blk)+p5*dlta_lon)
399 end do
400 do j_blk = 1,3
401 s_jj(j_blk) = nearest_j_north(lat_blk(j_blk)-p5*dlta_lat)
402 e_jj(j_blk) = nearest_j_south(lat_blk(j_blk)+p5*dlta_lat)
403 end do
404
405
406 ! Calculate lat/lon relevant to each "coarse grid" block
407 do i_blk = 1,3
408
409 ! "Shave" blocks on north or south due to proximity to poles
410 ! if necessary
411 j_blk = 1 ! southern row
412 ! Check for "shaved" block due to proximity to south pole
413 if ( (s_jj(j_blk).eq.-999).and.(e_jj(j_blk).ne.-999) ) then
414 s_jj(j_blk) = 1 ! southern boundary of shaved block
415 ! Reassign latitude of block center
416 lat_blk(j_blk) = p5*(lat1d_fine(1)+lat1d_fine(e_jj(j_blk)))
417 end if
418
419 j_blk = 2 ! center row
420 ! Check for "shaved" block due to proximity to south or north pole
421 ! Note: We're assuming e_jj(2) and s_jj(2) can't both be -999
422 if ( s_jj(j_blk).eq.-999 ) then
423 s_jj(j_blk) = 1 ! block shaved on the south
424 ! Reassign latitude of block center
425 lat_blk(j_blk) = p5*(lat1d_fine(1)+lat1d_fine(e_jj(j_blk)))
426 end if
427 if ( e_jj(j_blk).eq.-999 ) then
428 e_jj(j_blk) = dimy_fine ! block shaved on the north
429 ! Reassign latitude of block center
430 lat_blk(j_blk) = p5*(lat1d_fine(s_jj(j_blk))+lat1d_fine(dimy_fine))
431 end if
432
433 j_blk = 3 ! northern row
434 ! Check for "shaved" block due to proximity to north pole
435 if ( (e_jj(j_blk).eq.-999).and.(s_jj(j_blk).ne.-999) ) then
436 e_jj(j_blk) = dimy_fine ! northern boundary of shaved block
437 ! Reassign latitude of block center
438 lat_blk(j_blk) = p5*(lat1d_fine(s_jj(j_blk))+lat1d_fine(dimy_fine))
439 end if
440
441 end do
442
443
444 ! Calculate number of fine-grid points within center coarse block (2,2)
445 ! Check if center block straddles date line
446 if ( s_ii(2).gt.e_ii(2) ) then
447 ii_m = dimx_fine - s_ii(2) + 1 + e_ii(2)
448 else
449 ii_m = e_ii(2) - s_ii(2) + 1
450 end if
451 jj_m = e_jj(2) - s_jj(2) + 1
452
453
454 ! Assign values to "zs", which is the fine-grid surface topography field
455 ! that we will calculate statistics on, i.e, stddev, convexity, etc.
456 allocate (zs(ii_m,jj_m))
457
458 do jj = s_jj(2), e_jj(2)
459 jj_loc = jj - s_jj(2) + 1 ! local j-index (1 ... jj_m)
460 ! Check if block straddles the date line
461 if ( s_ii(2).gt.e_ii(2) ) then
462 do ii = s_ii(2), dimx_fine ! west of the date line
463 ii_loc = ii - s_ii(2) + 1 ! local i-index ( 1 ... ii_m)
464 zs(ii_loc,jj_loc) = hgt_m_fine(ii,jj)
465 end do
466 do ii = 1, e_ii(2) ! east of the date line
467 ii_loc = ii_loc + 1 ! local i-index ( 1 ... ii_m )
468 zs(ii_loc,jj_loc) = hgt_m_fine(ii,jj)
469 end do
470 else ! no crossing of the date line
471 do ii = s_ii(2), e_ii(2)
472 ii_loc = ii - s_ii(2) + 1 ! local i-index ( 1 ... ii_m)
473 zs(ii_loc,jj_loc) = hgt_m_fine(ii,jj)
474 end do
475 end if
476 end do
477
478
479
480 !
481 ! Finally, we can now calculate the topographic statistics fields needed
482 ! for the gravity wave drag scheme
483 !
484
485 ! Make sure statistics are zero if there is no terrain in the grid cell
486 ! Note: This is a proxy for a landmask
487 zs_accum = .false.
488 do jj = 1,jj_m
489 do ii = 1,ii_m
490 if ( abs(zs(ii,jj)).gt.1.e-1 ) zs_accum = .true.
491 end do
492 end do
493 if ( .not.zs_accum ) then ! no terrain in the grid cell
494 std_dev(cell_count) = 0._real_kind
495 convexity(cell_count) = 0._real_kind
496 oa1(cell_count) = 0._real_kind
497 oa2(cell_count) = 0._real_kind
498 oa3(cell_count) = 0._real_kind
499 oa4(cell_count) = 0._real_kind
500 ol1(cell_count) = 0._real_kind
501 ol2(cell_count) = 0._real_kind
502 ol3(cell_count) = 0._real_kind
503 ol4(cell_count) = 0._real_kind
504 deallocate(zs)
505 cycle ! move on to next (coarse) grid cell
506 end if
507
508
509 !
510 ! Calculate standard deviation of subgrid-scale terrain height
511 !
512
513 ! Calculate mean height
514 sum2 = 0._real_kind
515 nfinepoints = ii_m*jj_m
516 do jj = 1,jj_m
517 do ii = 1,ii_m
518 sum2 = sum2 + zs(ii,jj)
519 end do
520 end do
521 zs_mean = sum2 / real(nfinepoints,real_kind)
522
523 ! Calculate standard deviation
524 sum2 = 0._real_kind
525 do jj = 1,jj_m
526 do ii = 1,ii_m
527 sum2 = sum2 + ( zs(ii,jj) - zs_mean )**2
528 end do
529 end do
530 std_dev(cell_count) = sqrt( sum2/real(nfinepoints,real_kind) )
531
532 !
533 ! Calculate convexity of sub-grid-scale terrain
534 !
535
536 sum2 = 0._real_kind
537 sum4 = 0._real_kind
538 do jj = 1,jj_m
539 do ii = 1,ii_m
540 sum2 = sum2 + ( zs(ii,jj) - zs_mean )**2
541 sum4 = sum4 + ( zs(ii,jj) - zs_mean )**4
542 end do
543 end do
544
545 var = sum2 / real(nfinepoints,real_kind)
546 if ( abs(var) < 1.0e-05_real_kind ) then
547 convexity(cell_count) = 0._real_kind
548 else
549 convexity(cell_count) = min( sum4 / ( var**2 * &
550 real(nfinepoints,real_kind) ), max_convexity )
551 end if
552
553
554 !
555 ! Calculate orographic asymmetries
556 !
557
558 ! OA1 -- orographic asymmetry in West direction
559 nu = 0
560 nd = 0
561 do jj = 1,jj_m
562 if(mod(ii_m,2).eq.0.) then
563 do ii = 1,ii_m/2 ! left half of box
564 if ( zs(ii,jj) > zs_mean ) nu = nu + 1
565 end do
566 else
567 do ii = 1,ii_m/2+1 ! left half of box
568 if ( zs(ii,jj) > zs_mean ) nu = nu + 1
569 end do
570 endif
571 do ii = ii_m/2 + 1, ii_m ! right half of box
572 if ( zs(ii,jj) > zs_mean ) nd = nd + 1
573 end do
574 end do
575 if ( nu + nd > 0 ) then
576 oa1(cell_count) = real((nu - nd),real_kind) / &
577 real((nu + nd),real_kind)
578 else
579 oa1(cell_count) = 0._real_kind
580 end if
581
582 ! OA2 -- orographic asymmetry in South direction
583 nu = 0
584 nd = 0
585 if(mod(jj_m,2).eq.0.) then
586 do jj = 1,jj_m/2 ! bottom half of box
587 do ii = 1,ii_m
588 if ( zs(ii,jj) > zs_mean ) nu = nu + 1
589 end do
590 end do
591 else
592 do jj = 1,jj_m/2+1 ! bottom half of box
593 do ii = 1,ii_m
594 if ( zs(ii,jj) > zs_mean ) nu = nu + 1
595 end do
596 end do
597 endif
598 do jj = jj_m/2 + 1,jj_m ! top half of box
599 do ii = 1, ii_m
600 if ( zs(ii,jj) > zs_mean ) nd = nd + 1
601 end do
602 end do
603 if ( nu + nd > 0 ) then
604 oa2(cell_count) = real((nu - nd),real_kind) / &
605 real((nu + nd),real_kind)
606 else
607 oa2(cell_count) = 0._real_kind
608 end if
609
610 ! OA3 -- orographic asymmetry in South-West direction
611 nu = 0
612 nd = 0
613 ratio = real(jj_m,real_kind)/real(ii_m,real_kind)
614 do jj = 1,jj_m
615 do ii = 1,ii_m
616 if ( nint(real(ii,real_kind)*ratio) <= (jj_m - jj + 1) ) then
617 ! south-west half of box
618 if ( zs(ii,jj) > zs_mean ) nu = nu + 1
619 endif
620 if ( nint(real(ii,real_kind)*ratio) >= (jj_m - jj + 1) ) then
621 ! north-east half of box
622 if ( zs(ii,jj) > zs_mean ) nd = nd + 1
623 end if
624 end do
625 end do
626 if ( nu + nd > 0 ) then
627 oa3(cell_count) = real((nu - nd),real_kind) / &
628 real((nu + nd),real_kind)
629 else
630 oa3(cell_count) = 0._real_kind
631 end if
632
633 ! OA4 -- orographic asymmetry in North-West direction
634 nu = 0
635 nd = 0
636 ratio = real(jj_m,real_kind)/real(ii_m,real_kind)
637 do jj = 1,jj_m
638 do ii = 1,ii_m
639 if ( nint(real(ii,real_kind)*ratio) <= jj ) then
640 ! north-west half of box
641 if ( zs(ii,jj) > zs_mean ) nu = nu + 1
642 end if
643 if ( nint(real(ii,real_kind)*ratio) >= jj ) then
644 ! south-east half of box
645 if ( zs(ii,jj) > zs_mean ) nd = nd + 1
646 end if
647 end do
648 end do
649 if ( nu + nd > 0 ) then
650 oa4(cell_count) = real((nu - nd),real_kind) / &
651 real((nu + nd),real_kind)
652 else
653 oa4(cell_count) = 0._real_kind
654 end if
655
656
657 !
658 ! Calculate orographic effective lengths
659 !
660
661 ! OL1 -- orographic effective length for Westerly flow
662 nw = 0
663 nt = 0
664 do jj = max(jj_m/4,1), 3*jj_m/4
665 ! within central east-west band of box
666 do ii = 1, ii_m
667 if ( zs(ii,jj) > zs_mean ) nw = nw + 1
668 nt = nt + 1
669 end do
670 end do
671 if ( nt /= 0 ) then
672 ol1(cell_count) = real(nw,real_kind) / real(nt,real_kind)
673 else
674 ol1(cell_count) = 0._real_kind
675 end if
676
677 ! OL2 -- orographic effective length for Southerly flow
678 nw = 0
679 nt = 0
680 do jj = 1, jj_m
681 do ii = max(ii_m/4,1), 3*ii_m/4
682 ! within central north-south band of box
683 if ( zs(ii,jj) > zs_mean ) nw = nw + 1
684 nt = nt + 1
685 end do
686 end do
687 if ( nt /= 0 ) then
688 ol2(cell_count) = real(nw,real_kind) / real(nt,real_kind)
689 else
690 ol2(cell_count) = 0._real_kind
691 end if
692
693 ! OL3 -- orographic effective length for South-Westerly flow
694 nw = 0
695 nt = 0
696 do jj = 1, jj_m/2
697 do ii = 1, ii_m/2
698 if ( zs(ii,jj) > zs_mean ) nw = nw + 1
699 nt = nt + 1
700 end do
701 end do
702 do jj = jj_m/2+1, jj_m
703 do ii = ii_m/2+1, ii_m
704 if ( zs(ii,jj) > zs_mean ) nw = nw + 1
705 nt = nt + 1
706 end do
707 end do
708 if ( nt /= 0 ) then
709 ol3(cell_count) = real(nw,real_kind) / real(nt,real_kind)
710 else
711 ol3(cell_count) = 0._real_kind
712 end if
713
714 ! OL4 -- orographic effective length for North-Westerly flow
715 nw = 0
716 nt = 0
717 do jj = jj_m/2+1, jj_m
718 do ii = 1, ii_m/2
719 if ( zs(ii,jj) > zs_mean ) nw = nw + 1
720 nt = nt + 1
721 end do
722 end do
723 do jj = 1, jj_m/2
724 do ii = ii_m/2+1, ii_m
725 if ( zs(ii,jj) > zs_mean ) nw = nw + 1
726 nt = nt + 1
727 end do
728 end do
729 if ( nt /= 0 ) then
730 ol4(cell_count) = real(nw,real_kind) / real(nt,real_kind)
731 else
732 ol4(cell_count) = 0._real_kind
733 end if
734
735 deallocate (zs)
736
737 end do ! j = 1,dimY_FV3
738end do ! i = 1,dimX_FV3
